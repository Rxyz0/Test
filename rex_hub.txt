local canFile = (writefile and readfile and isfile) and true or false

local function saveSettings()
    if not canFile then return end
    local data
    if settings.saveEnabled then
        data = {
            saveEnabled = true,
            antiAfk = settings.antiAfk,
            showTimer = settings.showTimer,
            lastID = currentID,
            progress = claimedSuccessCount,
        }
    else
        data = { saveEnabled = false }
    end
    pcall(function()
        writefile(FILE_NAME, HttpService:JSONEncode(data))
    end)
end

local function loadSettings()
    if not canFile then return end
    local ok, raw = pcall(function()
        if isfile(FILE_NAME) then return readfile(FILE_NAME) end
    end)
    if not ok or not raw then return end

    local ok2, data = pcall(function() return HttpService:JSONDecode(raw) end)
    if not ok2 or type(data) ~= "table" then return end

    if data.saveEnabled == false then
        settings.saveEnabled = false
        return
    end
    if type(data.antiAfk) == "boolean" then settings.antiAfk = data.antiAfk end
    if type(data.showTimer) == "boolean" then settings.showTimer = data.showTimer end
    if type(data.lastID) == "number" then savedID = data.lastID end
    if type(data.progress) == "number" then savedProgress = math.clamp(math.floor(data.progress), 0, 9) end
end

loadSettings()

local function createToggle(par, y, text, initial, onChange)
    local row = make("Frame", {
        Size = UDim2.new(1, 0, 0, 34),
        Position = UDim2.new(0, 0, 0, y),
        BackgroundColor3 = COLOR_WHITE,
        BackgroundTransparency = 0.95,
        BorderSizePixel = 0,
    }, par)
    styleRow(row)

    local label = make("TextLabel", {
        Size = UDim2.new(1, -64, 1, 0),
        Position = UDim2.new(0, 12, 0, 0),
        BackgroundTransparency = 1,
        Text = text,
        TextColor3 = COLOR_TEXT,
        TextSize = 12,
        Font = Enum.Font.GothamMedium,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, row)

    local track = make("Frame", {
        Size = UDim2.new(0, 36, 0, 18),
        Position = UDim2.new(1, -48, 0.5, -9),
        BackgroundColor3 = COLOR_WHITE,
        BackgroundTransparency = 0.85,
        BorderSizePixel = 0,
    }, row)
    make("UICorner", { CornerRadius = UDim.new(1, 0) }, track)

    local knob = make("Frame", {
        Size = UDim2.new(0, 12, 0, 12),
        Position = UDim2.new(0, 3, 0.5, -6),
        BackgroundColor3 = Color3.fromRGB(160, 160, 160),
        BorderSizePixel = 0,
    }, track)
    make("UICorner", { CornerRadius = UDim.new(1, 0) }, knob)

    local hit = make("TextButton", {
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Text = "",
        ZIndex = 3,
    }, row)

    local state, locked, hint = initial and true or false, false, nil
    local tweenInfo = TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    local POS_ON = UDim2.new(1, -15, 0.5, -6)
    local POS_OFF = UDim2.new(0, 3, 0.5, -6)

    local function render(animate)
        local trackGoal, knobGoal
        if locked then
            trackGoal = { BackgroundTransparency = 0.93 }
            knobGoal = { Position = state and POS_ON or POS_OFF, BackgroundColor3 = Color3.fromRGB(80, 80, 80) }
        elseif state then
            trackGoal = { BackgroundTransparency = 0.1 }
            knobGoal = { Position = POS_ON, BackgroundColor3 = Color3.fromRGB(10, 10, 10) }
        else
            trackGoal = { BackgroundTransparency = 0.85 }
            knobGoal = { Position = POS_OFF, BackgroundColor3 = Color3.fromRGB(160, 160, 160) }
        end

        label.TextColor3 = locked and COLOR_MUTED or COLOR_TEXT
        label.Text = (locked and hint) and (text .. " (" .. hint .. ")") or text

        if animate then
            TweenService:Create(track, tweenInfo, trackGoal):Play()
            TweenService:Create(knob, tweenInfo, knobGoal):Play()
        else
            for k, v in pairs(trackGoal) do track[k] = v end
            for k, v in pairs(knobGoal) do knob[k] = v end
        end
    end

    hit.MouseButton1Click:Connect(function()
        if locked then return end
        state = not state
        render(true)
        if onChange then onChange(state) end
    end)

    render(false)

    local api = {}
    function api.get() return state end
    function api.set(v)
        state = v and true or false
        render(true)
    end
    function api.setLocked(v, h)
        locked = v
        hint = h
        render(true)
    end
    return api
end

local function selectTab(name)
    currentTab = name
    for n, t in pairs(tabs) do
        local active = (n == name)
        t.page.Visible = active
        t.button.BackgroundTransparency = active and 0.88 or 1
        t.button.TextColor3 = active and COLOR_TEXT or COLOR_MUTED
    end
    applySize()
end

            while ScreenGui.Parent do

                        if not isRunning or not ScreenGui.Parent then break end

                        local ok, res = pcall(function()
                            return claimRemote:InvokeServer(giftIndex, currentID)
                        end)

                        if ok and res == true then

                            GiftInputBox.Text = tostring(claimedSuccessCount)
                            saveSettings()
                            notify("Reward Claimed", "Claimed Gift #" .. giftIndex .. " for ID " .. currentID, 2)
                        end

                        task.wait(2)
                    end

                    if isRunning then
                        if claimedSuccessCount >= 9 then
                            setID(currentID + 1, true)
                            notify("Sequence Complete", "All 9 gifts claimed. Next ID: " .. tostring(currentID), 4)
                            task.wait(2)
                        else
                            setStatus("WAIT", COLOR_COOLDOWN)
                            notify("Status: Cooldown", "All available rewards claimed. Waiting...", 3)

                                if not isRunning or not ScreenGui.Parent then break end
                                task.wait(1)
                            end
                            if isRunning then setStatus("RUN", COLOR_TEXT) end
                        end
                    end
                else
                    task.wait(0.5)
                end
            end
        end)
    end
end

                delfile(FILE_NAME)
                cleared = true
            end
        end)
        notify("Saved Data", cleared and "Saved data cleared." or "No saved data found.", 2.5)
    end)
end

-- =================================================================
-- Drag (mouse & sentuh)
-- =================================================================
local dragging, dragInput, dragStart, startPos = false, nil, nil, nil

MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

local dragConn = UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

-- =================================================================
-- Animasi border + timer
-- =================================================================
local startTime = tick()
local lastSecond = -1

local renderConn = RunService.RenderStepped:Connect(function()
    local t = tick()
    StrokeGradient.Rotation = (t * 25) % 360

    local elapsed = math.floor(t - startTime)
    if elapsed ~= lastSecond then
        lastSecond = elapsed
        TimerLabel.Text = string.format("%02d:%02d:%02d",
            math.floor(elapsed / 3600),
            math.floor((elapsed % 3600) / 60),
            elapsed % 60)
    end
end)

-- =================================================================
-- Header buttons + cleanup
-- =================================================================
local function cleanup()
    isRunning = false
    pcall(function() idledConn:Disconnect() end)
    pcall(function() renderConn:Disconnect() end)
    pcall(function() dragConn:Disconnect() end)
    pcall(function() ScreenGui:Destroy() end)
    if env.AFKHubConn == idledConn then env.AFKHubConn = nil end
    env.RexHubCleanup = nil
end
env.RexHubCleanup = cleanup

MinimizeBtn.MouseButton1Click:Connect(function()
    isMinimized = not isMinimized
    ContentFrame.Visible = not isMinimized
    HeaderCover.Visible = not isMinimized
    MinimizeBtn.Text = isMinimized and "+" or "—"
    applySize()
end)

CloseBtn.MouseButton1Click:Connect(function()
    notify("Terminated", "Rex Hub closed. All features stopped.", 2)
    cleanup()
end)
